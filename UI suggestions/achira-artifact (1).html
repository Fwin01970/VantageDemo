Generated SQL

SELECT
    region,
    SUM(revenue) AS total_revenue,
    ROUND(
        (SUM(revenue) - LAG(SUM(revenue)) OVER (
            PARTITION BY region ORDER BY fiscal_year
        )) * 100.0
        / NULLIF(LAG(SUM(revenue)) OVER (
            PARTITION BY region ORDER BY fiscal_year
        ), 0),
        2
    ) AS yoy_growth
FROM analytics.regional_revenue
WHERE transaction_date BETWEEN
    '2026-01-01' AND '2026-08-15'
GROUP BY region, fiscal_year
ORDER BY total_revenue DESC
LIMIT 100;
